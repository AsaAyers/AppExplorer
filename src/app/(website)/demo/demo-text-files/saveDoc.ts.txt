import { emit } from "../bus/emit";

/**
 * Reading this code directly, it is 
 * not clear where this event will be
 * handled. If this is something that
 * people spend time looking for I will
 * put it on the board.
 * 
 * @AppExplorer
 */
export async function saveDoc(
  id: string,
  text: string,
) {
  emit("doc.saved", {
    id,
    bytes: text.length,
  });
}
